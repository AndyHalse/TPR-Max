// Helper the generator can use to fetch by id or category.
export function getFeaturesByIds(ids: string[]): TprFeature[] {
  return tprFeatures.filter((f) => ids.includes(f.id));
}

export function getFeaturesByCategory(
  category: TprFeature["category"]
): TprFeature[] {
  return tprFeatures.filter((f) => f.category === category);
}
