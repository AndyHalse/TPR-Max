TASK: Create a reusable plan-checking middleware and apply it to all
server routes that serve Pro or Max-only features. This enforces the
three-tier pricing — a Basic customer cannot access Pro/Max endpoints.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PART A — Create server/middleware/requirePlan.ts (new file)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Create this middleware function:

  import { Request, Response, NextFunction } from 'express';
  import { db } from '../db';
  import { customers } from '../../shared/schema';
  import { eq } from 'drizzle-orm';
  import { logger } from '../utils/logger';

  type Plan = 'tpr_basic' | 'tpr_pro' | 'tpr_max';

  const PLAN_HIERARCHY: Record<Plan, number> = {
    tpr_basic: 1,
    tpr_pro:   2,
    tpr_max:   3,
  };

  export function requirePlan(minimumPlan: Plan) {
    return async (req: Request, res: Response, next: NextFunction) => {
      try {
        const customerId = req.customerId;
        if (!customerId) {
          return res.status(401).json({ error: 'Authentication required' });
        }

        const [customer] = await db
          .select({ planType: customers.planType })
          .from(customers)
          .where(eq(customers.id, customerId))
          .limit(1);

        if (!customer) {
          return res.status(401).json({ error: 'Customer not found' });
        }

        const customerPlan = (customer.planType || 'tpr_basic') as Plan;
        const customerLevel = PLAN_HIERARCHY[customerPlan] ?? 1;
        const requiredLevel = PLAN_HIERARCHY[minimumPlan];

        if (customerLevel < requiredLevel) {
          return res.status(403).json({
            error: `This feature requires ${minimumPlan.replace('_', ' ').toUpperCase()}. Your current plan is ${customerPlan.replace('_', ' ').toUpperCase()}.`,
            upgradeRequired: true,
            currentPlan: customerPlan,
            requiredPlan: minimumPlan
          });
        }

        next();
      } catch (error) {
        logger.error('requirePlan middleware error:', error);
        next(error);
      }
    };
  }
